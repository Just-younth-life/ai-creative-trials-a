Agent 调度框架Agent Framework

灰度开关 + 工厂模式，把模型变成可替换零件

轻量化多模型调度框架：配置层、数据层、调度层、代理层、接入层解耦，用产品灰度开关控制推理策略与工具清洗逻辑，新增模型厂商只需扩展一个代理类。同一份代码既支持命令行调试，也支持 FastAPI 对外提供接口。

  • 10个源码文件
  • 3家模型厂商
  • 2种调用方式
  • 4层解耦结构

分层设计与接口契约

轻量化多模型调度框架

一、为什么要这一层

做 AI 原型最烦的不是接模型，是换模型。同一套业务逻辑，今天要跑 Azure，明天客户要求 Bedrock，后天想试国产厂商，如果每个模型都写一遍调用逻辑，需求一变就得改业务代码。

这个模块把「调哪家模型」变成配置项，把「怎么调」收进工厂。

二、四层结构

层 文件 职责
配置层 config_const.py / config_loader.py 灰度开关常量、配置加载校验、单例管理
数据层 schema.py / exceptions.py Pydantic 入参出参模型、统一业务异常
调度层 agent_core.py / prompt_templates.py 组装提示词、调用代理、回收工具调用与思考块
代理层 tool_provider.py 多模型代理工厂（Azure / Bedrock / Minimax）
接入层 main.py / api_server.py CLI 调试入口、FastAPI HTTP 入口

三、三个设计点

1. 灰度开关驱动行为
推理策略、工具清洗逻辑都挂在配置开关上。改行为不用改业务代码、不用重启服务，模拟真实产品的 A/B 灰度运营。

2. 工厂模式收敛差异
统一封装各厂商的代理类，新增一家只需要实现一个代理类并注册，调度层无感。

3. 强类型提前拦截
入参与配置都用 Pydantic 校验，非法配置在启动阶段就报错，而不是等到线上某个请求才炸。

四、怎么跑

pip install -r requirements.txt

python main.py          # 命令行本地调试，打印完整调度链路
python api_server.py    # 启动 FastAPI HTTP 服务

五、接口契约

POST /api/agent/chat

入参 类型 说明
user_query string 用户问题
session_id string | null 会话标识，缺省则新建
model_provider string azure / bedrock / minimax，默认 azure

返回 {code, msg, data}，data 内含 provider、content、tool_calls、thinking_block。异常统一走 AgentBaseError 分支，HTTP 400 携带异常类名，便于前端定位。

class AiCreativeAgent:
    def __init__(self):
        pass

    def _check_claude_version(self):
        """灰度：强制版本校验逻辑演示"""
        if get_gray_switch("oauth_force_claude_version_check"):
            mock_current_version = "claude-2.0"
            allow_version = "claude-3"
            if mock_current_version != allow_version:
                raise VersionCheckFailedError("Claude版本不匹配，请升级模型版本")

    async def run(self, req: AgentRequest) -> AgentResponse:
        # 1. 灰度分支：版本校验
        self._check_claude_version()

        # 2. 根据灰度开关动态组装prompt
        prompt = build_agent_prompt(

        )

        # 3. 获取对应厂商模型代理
        proxy = ModelProxyFactory.get_proxy(req.model_provider)

        # 4. 请求大模型
        resp = await proxy.chat_completion(prompt)
        return resp

class BaseModelProxy:
    """第三方模型代理基类"""
    async def chat_completion(self, prompt: str) -> AgentResponse:
        raise NotImplementedError


class AzureProxy(BaseModelProxy):
    async def chat_completion(self, prompt: str) -> AgentResponse:
        # 此处替换真实azure sdk请求逻辑
        return AgentResponse(content="azure model reply", provider="azure")


class MinimaxProxy(BaseModelProxy):
    async def chat_completion(self, prompt: str) -> AgentResponse:
        return AgentResponse(content="minimax model reply", provider="minimax")


class BedrockProxy(BaseModelProxy):
    async def chat_completion(self, prompt: str) -> AgentResponse:
        return AgentResponse(content="bedrock model reply", provider="bedrock")


class ModelProxyFactory:
    _mapping: Dict[str, BaseModelProxy] = {
        "azure": AzureProxy(),
        "minimax": MinimaxProxy(),
        "bedrock": BedrockProxy()
    }

    @classmethod
    def get_proxy(cls, provider: str) -> BaseModelProxy:
        gray_cfg = load_gray_config()
        if not is_provider_supported(provider, gray_cfg.third_party_proxy_support):
            raise UnsupportedProviderError(f"不支持的模型厂商: {provider}")
        instance = cls._mapping.get(provider.lower())
        if not instance:
            raise ModelProxyError("代理实例不存在")
        return instance

def load_gray_config() -> GrayConfigSchema:
    """加载并校验全局灰度配置，单例缓存"""
    global _GRAY_CONFIG
    if _GRAY_CONFIG is not None:
        return _GRAY_CONFIG

    adapter = TypeAdapter(GrayConfigSchema)
    _GRAY_CONFIG = adapter.validate_python(PRODUCT_GRAY_CONFIG)
    return _GRAY_CONFIG

def get_gray_switch(key: str):
    """快速读取单个灰度配置"""
    cfg = load_gray_config()
    return getattr(cfg, key)

TOOL_STRIP_PROMPT = trim_extra_whitespace("""

""")


def build_agent_prompt(user_query: str, enable_thinking_downgrade: bool, enable_strict_tool_clean: bool) -> str:
    """根据灰度开关动态组装Prompt"""
    segments = [SYSTEM_PROMPT]
    if enable_thinking_downgrade:
        segments.append(THINKING_DOWNGRADE_PROMPT)
    if enable_strict_tool_clean:
        segments.append(TOOL_STRIP_PROMPT)
    segments.append(f"用户问题：{user_query}")
    return "\n\n".join(segments)

from pydantic import BaseModel, Field

import json
from typing import Any


def safe_json_loads(text: str) -> Any:
    """安全解析JSON，避免直接json.loads抛错"""
    try:
        return json.loads(text)
    except Exception:
        return None


def is_provider_supported(provider: str, support_list: list[str]) -> bool:
    """校验模型厂商是否在支持列表内"""
    return provider.lower() in [p.lower() for p in support_list]


def trim_extra_whitespace(text: str) -> str:
    """清理多余换行与空格，用于prompt预处理"""
    return "\n".join(line.strip() for line in text.splitlines() if line.strip())

import asyncio
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

from agent.agent_core import AiCreativeAgent
from agent.schema import AgentRequest
from agent.exceptions import AgentBaseError

# 初始化FastAPI应用
app = FastAPI(title="AI-Creative-Trials Agent Service", version="1.0")
agent = AiCreativeAgent()

# HTTP接收请求模型
class AgentHttpReq(BaseModel):
    user_query: str
    session_id: str | None = None
    model_provider: str = "azure"


@app.post("/api/agent/chat")
async def agent_chat(req: AgentHttpReq):
    """Agent对话接口"""
    try:
        # HTTP入参转换为内部领域模型
        inner_req = AgentRequest(
            user_query=req.user_query,
            session_id=req.session_id,
            model_provider=req.model_provider
        )
        resp = await agent.run(inner_req)
        return {
            "code": 0,
            "msg": "success",
            "data": resp.model_dump()
        }
    except AgentBaseError as e:
        raise HTTPException(status_code=400, detail=f"{type(e).__name__}: {str(e)}")
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Server Error: {str(e)}")
if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
